function sendEmails() { var sheet = SpreadsheetApp.getActiveSheet(); //get spreadsheet var dataRange = sheet.getRange(2, 1, 50, 7); //get all data from (row2,column1) to (row50,column7) var rows = dataRange.getValues(); //split the data back up into rows, starting with (0,0) now for (var i=0; i<50; i=i+1) { //For each row we took: var thisRow = rows[i]; //call it "thisRow" if (thisRow[5] && !thisRow[5].includes("***") && thisRow[5].includes("@")) { //if thisRow's sixth column (thisRow[5]) is an email address that doesn't already include "***", then: //create a subject line using the Name from thisRow's fifth column (thisRow[4]) var subject = thisRow[4] + ', get a look on it' ; //create a message (with line spaces '\n') using the Name from thisRow's fifth column (thisRow[4]) var message = 'Hey ' + thisRow[4] + ',\n\n' + 'how everything is going on?' + '\n\n' ; var emailAddress = thisRow[5]; //capture the email address from thisRow's sixth column (thisRow[5]) var message = 'Hey ' + thisRow[4] + ',\n\n' + 'how everything is going on?' + '\n\n' + 'Hope everything is well. ' + '\n\n' + 'Your favourite color is ' + thisRow[6] + '.\n\n' + 'Sincerely,\n' + 'Samanta'; MailApp.sendEmail(emailAddress, subject, message); //send an email using the above details console.log(emailAddress) sheet.getRange(i+2, 6).setValue(emailAddress.concat('***')); //add '***' to the end of the email address, so an email won't be re-sent to that address again in the future SpreadsheetApp.flush(); //update that cell immediately } } }
